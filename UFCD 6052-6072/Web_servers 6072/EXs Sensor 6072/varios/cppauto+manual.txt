#include <WiFi.h>
#include <WebServer.h>

// Substitua pelas suas credenciais de rede Wi-Fi
const char* ssid = "IEFP-Formacao";
const char* password = "";

// Atribui variáveis de saída aos pinos GPIO
// Mudei os pinos dos LEDs para 4 e 12 conforme o seu último código.
// Certifique-se de que estes pinos são adequados para LEDs (não são pinos de entrada apenas, etc.).
const int output26 = 4; // LED 1
const int output27 = 12; // LED 2

// --- Configuração do LDR ---
const int LDR_PIN = 34; // Pino GPIO para o LDR (entrada analógica)
const int LDR_LIMIT = 2000; // Limite para a leitura do LDR (ajuste este valor!)
                            // A leitura analógica do ESP32 vai de 0 (escuro) a 4095 (claro).
                            // Ajuste LDR_LIMIT com base na sua luz ambiente e resistor.

// Variável para armazenar a leitura atual do LDR
int ldrValue = 0;

// Variáveis para controlar o tempo da atualização da leitura do sensor LDR
unsigned long lastLDRReadMillis = 0;
const long ldrReadInterval = 500; // Ler o LDR a cada 500 milissegundos

// --- Modo de Operação ---
enum OperatingMode {
  AUTOMATIC_MODE,
  MANUAL_MODE
};
OperatingMode currentMode = AUTOMATIC_MODE; // Começa no modo automático por padrão

// Variáveis para armazenar o estado manual dos LEDs (se estiver no modo manual)
String output26ManualState = "off";
String output27ManualState = "off";

// Cria um objeto do servidor web na porta 80
WebServer server(80);

// --- Funções Auxiliares ---

// Função para ler o valor do LDR
int readLDR() {
  return analogRead(LDR_PIN);
}

// --- Handlers para o Modo Manual ---
void handleSet26() {
  if (server.hasArg("state")) {
    String state = server.arg("state");
    if (state == "on") {
      digitalWrite(output26, HIGH);
      output26ManualState = "on";
    } else if (state == "off") {
      digitalWrite(output26, LOW);
      output26ManualState = "off";
    }
  }
  server.sendHeader("Location", "/"); // Redireciona de volta para a página principal
  server.send(303); // Resposta "See Other"
}

void handleSet27() {
  if (server.hasArg("state")) {
    String state = server.arg("state");
    if (state == "on") {
      digitalWrite(output27, HIGH);
      output27ManualState = "on";
    } else if (state == "off") {
      digitalWrite(output27, LOW);
      output27ManualState = "off";
    }
  }
  server.sendHeader("Location", "/");
  server.send(303);
}

// --- Handlers para Mudar de Modo ---
void handleAutomaticMode() {
  currentMode = AUTOMATIC_MODE;
  // Quando muda para automático, desliga os LEDs para que o LDR possa controlá-los do zero
  digitalWrite(output26, LOW);
  digitalWrite(output27, LOW);
  server.sendHeader("Location", "/");
  server.send(303);
}

void handleManualMode() {
  currentMode = MANUAL_MODE;
  // Quando muda para manual, desliga os LEDs para que o utilizador possa controlá-los
  digitalWrite(output26, LOW);
  digitalWrite(output27, LOW);
  output26ManualState = "off"; // Reseta o estado manual
  output27ManualState = "off"; // Reseta o estado manual
  server.sendHeader("Location", "/");
  server.send(303);
}

// --- Função de Configuração (executada uma vez no início) ---
void setup() {
  Serial.begin(115200);
  Serial.println("\nIniciando ESP32 Web Server com Controlo de Iluminação...");

  pinMode(output26, OUTPUT);
  pinMode(output27, OUTPUT);
  digitalWrite(output26, LOW);
  digitalWrite(output27, LOW);

  Serial.print("Conectando ao WiFi ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);

  int attempts = 0;
  const int maxAttempts = 5;

  while (WiFi.status() != WL_CONNECTED && attempts < maxAttempts) {
    delay(1000);
    Serial.print(".");
    attempts++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi Conectado!");
    Serial.print("Endereço IP: ");
    Serial.println(WiFi.localIP());

    // Configura as rotas do servidor web
    server.on("/", handleRoot);
    server.on("/automatic", handleAutomaticMode);
    server.on("/manual", handleManualMode);
    server.on("/set26", handleSet26); // Rota para controlo manual do GPIO 26
    server.on("/set27", handleSet27); // Rota para controlo manual do GPIO 27

    server.begin();
    Serial.println("Servidor Web HTTP iniciado.");

    // Força uma primeira leitura do LDR no início
    ldrValue = readLDR();
    Serial.print("Leitura LDR inicial: "); Serial.println(ldrValue);

  } else {
    Serial.println("\nNão foi possível estabelecer conexão Wi-Fi após 5 tentativas.");
    Serial.println("Verifique as credenciais ou o alcance da rede.");
  }
}

// --- Função Principal do Loop (executada repetidamente) ---
void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    server.handleClient();

    // Lógica para ler o LDR e controlar os LEDs APENAS no modo AUTOMÁTICO
    if (currentMode == AUTOMATIC_MODE) {
      unsigned long currentMillis = millis();
      if (currentMillis - lastLDRReadMillis >= ldrReadInterval) {
        lastLDRReadMillis = currentMillis;

        ldrValue = readLDR();
        Serial.print("LDR Value (Auto Mode): ");
        Serial.println(ldrValue);

        if (ldrValue < LDR_LIMIT) { // Se estiver mais escuro que o limite
          digitalWrite(output26, HIGH); // Liga o LED 1 (GPIO 4)
          digitalWrite(output27, LOW);  // Desliga o LED 2 (GPIO 12)
          Serial.println("Ambiente escuro: LED 4 LIGADO.");
        } else { // Se estiver mais claro que o limite
          digitalWrite(output26, LOW);  // Desliga o LED 1 (GPIO 4)
          digitalWrite(output27, HIGH); // Liga o LED 2 (GPIO 12)
          Serial.println("Ambiente claro: LED 12 LIGADO.");
        }
      }
    }
    // No modo MANUAL, os LEDs são controlados pelos handlers /set26 e /set27
    // e a lógica do LDR é ignorada.
  }
}
